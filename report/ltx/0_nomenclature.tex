\begin{tabular}{ll}

  $b$ & Turbine rotor blade height (\si{\milli\meter}) \\
  $b_1$, $b_2$ & Pump blade axial width at stations 1, 2 (\si{\milli\meter}) \\
  $b_c$, $h_c$ & Annular casing width, radial depth (\si{\milli\meter}) \\
  $c$ & Absolute velocity (\si{\meter\per\second}) \\
  $c_p$ & Specific heat at constant pressure (\si{\joule\per\kilogram\per\kelvin}) \\
  $c_s$ & Spouting velocity (\si{\meter\per\second}) \\
  $C_H$, $K_1$ & Lock/Busemann head-correction constants ($-$) \\
  $d_0 \ldots d_4$ & Pump station diameters (\si{\milli\meter}) \\
  $d_m$ & Turbine mean (pitch) diameter (\si{\milli\meter}) \\
  $f$ & Friction coefficient ($-$) \\
  $F_R$ & Radial thrust (\si{\newton}) \\
  $g$ & Gravitational acceleration (\si{\meter\per\square\second}) \\
  $H$ & Pump head (\si{\meter}); $H_{th}$ theoretical (Euler) head; $H_3$ head at diffuser throat \\
  $h$ & Specific enthalpy (\si{\joule\per\kilogram}); $\Delta h_s$ isentropic drop \\
  $k_d$ & Barske diffuser kinetic-recovery coefficient ($-$) \\
  $k_N$, $k_R$ & Nozzle, rotor velocity coefficients (Weiss) ($-$) \\
  $k_0$ & Lock shutoff head constant ($-$) \\
  $K_R$ & Radial thrust coefficient ($-$) \\
  $M$ & Mach number; $M_{w3}$ rotor-inlet relative Mach ($-$) \\
  $\dot{m}$ & Mass flow rate (\si{\kilogram\per\second}) \\
  $n$ & Shaft speed (\si{\rpm}) \\
  $n_q$ & Specific speed $n\sqrt{Q}/H^{0.75}$ (rpm, \si{\cubic\meter\per\second}, m) \\
  $\textrm{NPSH}_a$, $\textrm{NPSH}_3$ & Available NPSH; NPSH at 3\,\% head drop (\si{\meter}) \\
  $p$ & Static pressure (\si{\pascal}); $p_{01}$, $T_{01}$ total state at station 1; $p_v$ vapour pressure \\
  $P$ & Power (\si{\watt}); $P_{RR}$ disc friction/churning; $P_w$ ventilation (windage) power \\
  $q$ & Dynamic pressure $\tfrac{1}{2}\rho c^2$ (\si{\pascal}) \\
  $Q$ & Volumetric flow rate (\si{\cubic\meter\per\second}); $Q_L$ leakage flow \\
  $r$ & Radius (\si{\meter}); recovery factor in \cref{eq:recovery_temperature} only \\
  $R$ & Specific gas constant (\si{\joule\per\kilogram\per\kelvin}) \\
  $Re_u$ & Peripheral Reynolds number $u_2 r_2/\nu_l$ ($-$) \\
  $s_{ax}$ & Impeller--casing axial clearance (\si{\milli\meter}) \\
  $T$ & Temperature (\si{\kelvin}) \\
  $u$ & Blade (peripheral) speed (\si{\meter\per\second}) \\
  $w$ & Relative velocity (\si{\meter\per\second}) \\
  $Y$ & Specific work (\si{\joule\per\kilogram}) \\
  $z$ & Blade count ($-$) \\
  $\alpha$ & Absolute flow angle from circumferential (\si{\degree}) \\
  $\beta$ & Relative flow / blade angle from circumferential (\si{\degree}) \\
  $\gamma$ & Heat capacity ratio ($-$) \\
  $\varepsilon$ & Degree of (partial) admission ($-$) \\
  $\zeta_D$ & Diffuser-nozzle loss coefficient ($-$) \\
  $\eta$ & Efficiency: pump overall $\rho g Q H/(\tau\omega)$; $\eta_h$ hydraulic; $\eta_{ts}$ turbine total-to-static \\
  $\mu$ & Dynamic viscosity (always with a state subscript); Mach angle within the MoC set only \\
  $\nu$ & Prandtl--Meyer angle (\si{\degree}); kinematic viscosity is written $\mu/\rho$ or $\nu_l$ \\
  $\rho$ & Density (\si{\kilogram\per\cubic\meter}); $\rho_b$ blade material density \\
  $\sigma$ & Slip factor (Wiesner), \S\ref{sec:theory} only ($-$) \\
  $\tau$ & Shaft torque (\si{\newton\meter}) \\
  $\phi$ & Pump flow coefficient $Q/(\pi d_2 b_2 u_2)$ ($-$) \\
  $\psi$ & Pump head coefficient $2gH/u_2^2$ ($-$) \\
  $\omega$ & Shaft angular velocity (\si{\radian\per\second}); $\omega_s$ nondimensional specific speed \\
\end{tabular}
